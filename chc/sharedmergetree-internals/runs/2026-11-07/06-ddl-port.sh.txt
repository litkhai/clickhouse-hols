-- 1. OSS DDL often names the cluster through a macro. Cloud defines no {cluster} macro.
CREATE TABLE smt_lab.p1 ON CLUSTER '{cluster}' (id UInt64) ENGINE = ReplicatedMergeTree ORDER BY id
  → fails as expected: No macro 'cluster' in config while processing substitutions in '{cluster}' at '1' or macro is not supported here. 

-- 2. Explicit Keeper path and replica name, as written for OSS: rejected in the Shared database
CREATE TABLE smt_lab.p2 (id UInt64) ENGINE = ReplicatedMergeTree('/clickhouse/tables/{shard}/p2', '{replica}') ORDER BY id
  → fails as expected: It's not allowed to specify explicit zookeeper_path and replica_name for ReplicatedMergeTree arguments in Replicated database. If you really want to specify them explicitly, enable setting database_replicated_allow_replicated_engine_arguments.. 

-- 3. The same with ON CLUSTER default: the cluster name resolves, the engine arguments still do not
CREATE TABLE smt_lab.p3 ON CLUSTER default (id UInt64) ENGINE = ReplicatedMergeTree('/clickhouse/tables/{shard}/p3', '{replica}') ORDER BY id
  → fails as expected: It's not allowed to specify explicit zookeeper_path and replica_name for ReplicatedMergeTree arguments in Replicated database. If you really want to specify them explicitly, enable setting database_replicated_allow_replicated_engine_arguments.. 

-- 4. What works: drop the arguments (and ON CLUSTER, or keep it as 'default')
CREATE TABLE smt_lab.p4 (id UInt64) ENGINE = ReplicatedMergeTree ORDER BY id
  → OK
CREATE TABLE smt_lab.p5 ON CLUSTER default (id UInt64, v UInt64) ENGINE = ReplicatedReplacingMergeTree(v) ORDER BY id
  → OK
CREATE TABLE smt_lab.p6 (id UInt64, v UInt64) ENGINE = AggregatingMergeTree ORDER BY id
  → OK
