=================== 01-setup.sql
-- 01-setup.sql: a raw table, a 1-minute rollup fed by a materialized view on the raw table,
-- and a 10-minute rollup fed by a materialized view on the 1-minute rollup (cascading).
--
-- The rollup holds the same distinct-user count nine times (one column per state function and
-- input type) so that system.parts_columns can give each one's size side by side. A real rollup
-- would hold one of them. Every rollup column is pinned to CODEC(LZ4) so the compressed sizes
-- do not depend on the build's default codec; uncompressed sizes are the state bytes themselves.

CREATE DATABASE IF NOT EXISTS mv_uniq;
-- Raw events: page views. The same user appears as UInt32, UInt64 and a 32-character String.
CREATE TABLE mv_uniq.events
(
    ts         DateTime('UTC')        CODEC(LZ4),
    kind       LowCardinality(String) CODEC(LZ4),
    page       UInt32                 CODEC(LZ4),
    user_u32   UInt32                 CODEC(LZ4),
    user_u64   UInt64                 CODEC(LZ4),
    user_str   String                 CODEC(LZ4),
    latency_ms Float32                CODEC(LZ4)
)
ENGINE = MergeTree
ORDER BY ts;
CREATE TABLE mv_uniq.rollup_1m
(
    bucket       DateTime('UTC')                           CODEC(LZ4),
    kind         LowCardinality(String)                    CODEC(LZ4),
    page         UInt32                                    CODEC(LZ4),
    events       SimpleAggregateFunction(sum, UInt64)      CODEC(LZ4),
    latency_sum  SimpleAggregateFunction(sum, Float64)     CODEC(LZ4),
    u_uniq_str   AggregateFunction(uniq, String)           CODEC(LZ4),
    u_uniq_u64   AggregateFunction(uniq, UInt64)           CODEC(LZ4),
    u_uniq_u32   AggregateFunction(uniq, UInt32)           CODEC(LZ4),
    u_comb_u64   AggregateFunction(uniqCombined, UInt64)   CODEC(LZ4),
    u_comb12_u64 AggregateFunction(uniqCombined(12), UInt64) CODEC(LZ4),
    u_hll12_u64  AggregateFunction(uniqHLL12, UInt64)      CODEC(LZ4),
    u_exact_str  AggregateFunction(uniqExact, String)      CODEC(LZ4),
    u_exact_u64  AggregateFunction(uniqExact, UInt64)      CODEC(LZ4),
    u_exact_u32  AggregateFunction(uniqExact, UInt32)      CODEC(LZ4)
)
ENGINE = AggregatingMergeTree
ORDER BY (bucket, kind, page)
SETTINGS min_bytes_for_wide_part = 0;
CREATE MATERIALIZED VIEW mv_uniq.mv_1m TO mv_uniq.rollup_1m AS
SELECT
    toStartOfMinute(ts)              AS bucket,
    kind,
    page,
    count()                          AS events,
    sum(toFloat64(latency_ms))       AS latency_sum,
    uniqState(user_str)              AS u_uniq_str,
    uniqState(user_u64)              AS u_uniq_u64,
    uniqState(user_u32)              AS u_uniq_u32,
    uniqCombinedState(user_u64)      AS u_comb_u64,
    uniqCombinedState(12)(user_u64)  AS u_comb12_u64,
    uniqHLL12State(user_u64)         AS u_hll12_u64,
    uniqExactState(user_str)         AS u_exact_str,
    uniqExactState(user_u64)         AS u_exact_u64,
    uniqExactState(user_u32)         AS u_exact_u32
FROM mv_uniq.events
GROUP BY bucket, kind, page;
-- Same columns, codecs and settings as the 1-minute rollup.
CREATE TABLE mv_uniq.rollup_10m AS mv_uniq.rollup_1m
ENGINE = AggregatingMergeTree
ORDER BY (bucket, kind, page)
SETTINGS min_bytes_for_wide_part = 0;
-- Cascading: the source is the 1-minute rollup, so this view sees each block the 1-minute view
-- writes and merges its partial states into 10-minute states. The raw rows are aggregated once.
CREATE MATERIALIZED VIEW mv_uniq.mv_10m TO mv_uniq.rollup_10m AS
SELECT
    toStartOfTenMinutes(bucket)               AS bucket,
    kind,
    page,
    sum(events)                               AS events,
    sum(latency_sum)                          AS latency_sum,
    uniqMergeState(u_uniq_str)                AS u_uniq_str,
    uniqMergeState(u_uniq_u64)                AS u_uniq_u64,
    uniqMergeState(u_uniq_u32)                AS u_uniq_u32,
    uniqCombinedMergeState(u_comb_u64)        AS u_comb_u64,
    uniqCombinedMergeState(12)(u_comb12_u64)  AS u_comb12_u64,
    uniqHLL12MergeState(u_hll12_u64)          AS u_hll12_u64,
    uniqExactMergeState(u_exact_str)          AS u_exact_str,
    uniqExactMergeState(u_exact_u64)          AS u_exact_u64,
    uniqExactMergeState(u_exact_u32)          AS u_exact_u32
FROM mv_uniq.rollup_1m
GROUP BY bucket, kind, page;

=================== 03-state-size.sql
-- 03-state-size.sql: how big one state is, by state function, distinct values per group and input type.
-- length(CAST(state AS String)) is the serialized state, the bytes a rollup row stores for it.
-- Independent of the rollup tables; values come from numbers().

-- (a) Bytes per state by distinct values per group (input UInt64, sipHash64 of distinct integers).
SELECT
    n                                                        AS distinct_values,
    length(CAST(uniqState(v) AS String))                     AS uniq,
    length(CAST(uniqCombinedState(v) AS String))             AS uniqCombined,
    length(CAST(uniqCombinedState(12)(v) AS String))         AS uniqCombined_12,
    length(CAST(uniqHLL12State(v) AS String))                AS uniqHLL12,
    length(CAST(uniqExactState(v) AS String))                AS uniqExact
FROM
(
    SELECT n, sipHash64(number) AS v
    FROM numbers(1000000)
    ARRAY JOIN arrayFilter(x -> number < x, [1, 10, 30, 100, 300, 600, 700, 1000, 3000, 10000, 30000, 100000, 1000000]) AS n
)
GROUP BY n
ORDER BY n;

        [length(CAST(uniqState(nstr) AS String)), length(CAST(uniqCombinedState(nstr) AS String)),
         length(CAST(uniqCombinedState(12)(nstr) AS String)), length(CAST(uniqHLL12State(nstr) AS String)),

        [length(CAST(uniqState(u64) AS String)), length(CAST(uniqCombinedState(u64) AS String)),
         length(CAST(uniqCombinedState(12)(u64) AS String)), length(CAST(uniqHLL12State(u64) AS String)),
         length(CAST(uniqExactState(u64) AS String))] AS a_u64,
        [length(CAST(uniqState(u32) AS String)), length(CAST(uniqCombinedState(u32) AS String)),
         length(CAST(uniqCombinedState(12)(u32) AS String)), length(CAST(uniqHLL12State(u32) AS String)),

-- (c) Relative error against the true count: 10 groups per distinct count, disjoint values.
-- uniqExact is the control: its error must be 0.
SELECT fn, n AS distinct_values,
       round(avg(abs(est - n) / n) * 100, 3) AS mean_err_pct,
       round(max(abs(est - n) / n) * 100, 3) AS max_err_pct
FROM
(
    SELECT g, n,
           ['uniq', 'uniqCombined', 'uniqCombined(12)', 'uniqHLL12', 'uniqExact'] AS fns,
           [uniq(v), uniqCombined(v), uniqCombined(12)(v), uniqHLL12(v), uniqExact(v)] AS ests
    FROM
    (
        SELECT g, n, sipHash64(g, number) AS v
        FROM numbers(1000000)
        ARRAY JOIN arrayFilter(x -> number < x, [10, 100, 1000, 10000, 100000, 1000000]) AS n
        ARRAY JOIN range(10) AS g
    )
    GROUP BY g, n
)
ARRAY JOIN fns AS fn, ests AS est
GROUP BY fn, n

=================== 05-cascade-check.sql
-- 05-cascade-check.sql: is the 10-minute rollup built from the 1-minute rollup right?

-- Positive check: the cascaded 10-minute rollup against 10-minute groups computed straight from the
-- raw rows. Both sides must have the same groups and events, uniqExact and uniq must match exactly.
-- The approximate functions are compared too: a merged state need not give the same estimate as
-- one built from all rows at once.
SELECT
    count()                                   AS groups,
    countIf(r.bucket_ok = 0 OR d.bucket_ok = 0) AS one_side_only,
    countIf(r.events != d.events)             AS events_diff,
    countIf(r.exact != d.exact)               AS uniqExact_diff,
    countIf(r.uq != d.uq)                     AS uniq_diff,
    countIf(r.comb != d.comb)                 AS uniqCombined_diff,
    countIf(r.comb12 != d.comb12)             AS uniqCombined12_diff,
    countIf(r.hll12 != d.hll12)               AS uniqHLL12_diff,
    -- where uniqCombined(12) differs: the true distinct counts of those groups and the largest gap
    minIf(d.exact, r.comb12 != d.comb12)      AS comb12_diff_min_true,
    maxIf(d.exact, r.comb12 != d.comb12)      AS comb12_diff_max_true,
    maxIf(abs(toInt64(r.comb12) - toInt64(d.comb12)), r.comb12 != d.comb12) AS comb12_max_gap
FROM
(
    SELECT bucket, kind, page, 1 AS bucket_ok,
           sum(events)                          AS events,
           uniqExactMerge(u_exact_u64)          AS exact,
           uniqMerge(u_uniq_u64)                AS uq,
           uniqCombinedMerge(u_comb_u64)        AS comb,
           uniqCombinedMerge(12)(u_comb12_u64)  AS comb12,
           uniqHLL12Merge(u_hll12_u64)          AS hll12
    FROM mv_uniq.rollup_10m
    GROUP BY bucket, kind, page
) AS r
FULL OUTER JOIN
(
    SELECT toStartOfTenMinutes(ts) AS bucket, kind, page, 1 AS bucket_ok,
           count()                    AS events,
           uniqExact(user_u64)        AS exact,
           uniq(user_u64)             AS uq,
           uniqCombined(user_u64)     AS comb,
           uniqCombined(12)(user_u64) AS comb12,
           uniqHLL12(user_u64)        AS hll12
    FROM mv_uniq.events
    GROUP BY bucket, kind, page
) AS d USING (bucket, kind, page);

-- (b) Per page over the last hour, every state column of the 1-minute rollup against uniqExact(UInt64):
-- relative error, and how many of the exact Top 10 / Bottom 10 pages it returns (and in the same order).
WITH per_page AS
(
    SELECT page, sum(events) AS events,
           uniqExactMerge(u_exact_u64) AS exact,
           [uniqMerge(u_uniq_str), uniqMerge(u_uniq_u64), uniqMerge(u_uniq_u32),
            uniqCombinedMerge(u_comb_u64), uniqCombinedMerge(12)(u_comb12_u64), uniqHLL12Merge(u_hll12_u64),
            uniqExactMerge(u_exact_str), uniqExactMerge(u_exact_u32)] AS ests
    FROM mv_uniq.rollup_1m
    WHERE bucket >= '2026-11-01 02:00:00'
    GROUP BY page
)
SELECT fn, pages, mean_err_pct, max_err_pct,
       top10_same_pages, top10_same_order, bottom10_same_pages, bottom10_same_order
FROM
(
SELECT
    fn,
    count() AS pages,
    round(avg(abs(est - exact) / exact) * 100, 3) AS mean_err_pct,
    round(max(abs(est - exact) / exact) * 100, 3) AS max_err_pct,
    arraySlice(arrayMap(t -> t.2, arraySort(t -> (-toInt64(t.1), t.2), groupArray((est, page)))), 1, 10)   AS top_est,
    arraySlice(arrayMap(t -> t.2, arraySort(t -> (-toInt64(t.1), t.2), groupArray((exact, page)))), 1, 10) AS top_exact,
    arraySlice(arrayMap(t -> t.2, arraySort(t -> (toInt64(t.1), t.2), groupArrayIf((est, page), events >= 1000))), 1, 10)   AS bottom_est,
    arraySlice(arrayMap(t -> t.2, arraySort(t -> (toInt64(t.1), t.2), groupArrayIf((exact, page), events >= 1000))), 1, 10) AS bottom_exact,
    length(arrayIntersect(top_est, top_exact))       AS top10_same_pages,
    top_est = top_exact                              AS top10_same_order,
    length(arrayIntersect(bottom_est, bottom_exact)) AS bottom10_same_pages,
    bottom_est = bottom_exact                        AS bottom10_same_order

=================== 08-timing.sql
-- 08-timing.sql: the 06-topn.sql runs from system.query_log (run.sh runs 06 five times; tools/hol once).
-- read_from_disk = ProfileEvents ReadCompressedBytes (bytes read from compressed files, before decompression),
-- decompressed = CompressedReadBufferBytes (the same data after decompression). query_log read_bytes is the
-- in-memory size of the columns read (SelectedBytes), which for state columns is not their size on disk.

SYSTEM FLUSH LOGS;
SELECT
    replaceOne(log_comment, 'topn ', '')                              AS probe,
    count()                                                           AS runs,
    median(query_duration_ms)                                         AS median_ms,
    min(query_duration_ms)                                            AS min_ms,
    any(read_rows)                                                    AS read_rows,
    formatReadableSize(any(ProfileEvents['ReadCompressedBytes']))     AS read_from_disk,
    formatReadableSize(any(ProfileEvents['CompressedReadBufferBytes'])) AS decompressed,
    formatReadableSize(any(read_bytes))                               AS read_bytes_in_memory,
    formatReadableSize(max(memory_usage))                             AS peak_memory
FROM system.query_log
WHERE type = 'QueryFinish' AND log_comment LIKE 'topn %'
GROUP BY log_comment
ORDER BY min(event_time_microseconds);
