## 00-setup.sql
-- 00-setup.sql: database, a generator for hourly full snapshots, and the tables every later file uses.
--
-- The source re-delivers a FULL snapshot every hour into a plain MergeTree (for example through the
-- ClickHouse Kafka Connect Sink). Each row: record_key, 25 Nullable(String) business fields,
-- event_timestamp (changes on every delivery) and ingestion_timestamp (set when the row is written).

CREATE DATABASE IF NOT EXISTS snapdiff;
-- gen(n, v, sid): one full snapshot of n records.
--   v = 0  the base snapshot: keys 0 .. n-1
--   v = 1  the next snapshot: keys with n % 1000 < 3 are gone (deleted, 0.3 %),
--          keys n .. n + n/200 - 1 are added (new, 0.5 %),
--          f03 changes where n % 100 = 7 (1 %), and f06 goes from NULL to '' where n % 100000 = 11.
-- f06, f12, f18, f22 and f25 are NULL for about 10 % of the keys; event_timestamp moves on every snapshot.
CREATE OR REPLACE VIEW snapdiff.gen AS
SELECT
    {sid:DateTime} AS snapshot_id,
    concat('K', leftPad(toString(n), 10, '0')) AS record_key,
    CAST(concat('C', toString(cityHash64(n, 1) % 40)) AS Nullable(String)) AS f01,
    CAST(concat('C', toString(cityHash64(n, 2) % 40)) AS Nullable(String)) AS f02,
    CAST(if({v:UInt8} = 1 AND n % 100 = 7, concat(concat('C', toString(cityHash64(n, 3) % 40)), '_x'), concat('C', toString(cityHash64(n, 3) % 40))) AS Nullable(String)) AS f03,
    CAST(concat('C', toString(cityHash64(n, 4) % 40)) AS Nullable(String)) AS f04,
    CAST(concat('C', toString(cityHash64(n, 5) % 40)) AS Nullable(String)) AS f05,
    CAST(multiIf(n % 100000 = 11, if({v:UInt8} = 1, '', NULL), cityHash64(n, 6, 'null') % 10 = 0, NULL, concat('C', toString(cityHash64(n, 6) % 40))) AS Nullable(String)) AS f06,
    CAST(concat('C', toString(cityHash64(n, 7) % 40)) AS Nullable(String)) AS f07,
    CAST(concat('C', toString(cityHash64(n, 8) % 40)) AS Nullable(String)) AS f08,
    CAST(concat('C', toString(cityHash64(n, 9) % 40)) AS Nullable(String)) AS f09,
    CAST(concat('C', toString(cityHash64(n, 10) % 40)) AS Nullable(String)) AS f10,
    CAST(substring(hex(cityHash64(n, 11)), 1, 12) AS Nullable(String)) AS f11,
    CAST(if(cityHash64(n, 12, 'null') % 10 = 0, NULL, substring(hex(cityHash64(n, 12)), 1, 12)) AS Nullable(String)) AS f12,
    CAST(substring(hex(cityHash64(n, 13)), 1, 12) AS Nullable(String)) AS f13,
    CAST(substring(hex(cityHash64(n, 14)), 1, 12) AS Nullable(String)) AS f14,
    CAST(substring(hex(cityHash64(n, 15)), 1, 12) AS Nullable(String)) AS f15,
    CAST(substring(hex(cityHash64(n, 16)), 1, 12) AS Nullable(String)) AS f16,
    CAST(substring(hex(cityHash64(n, 17)), 1, 12) AS Nullable(String)) AS f17,
    CAST(if(cityHash64(n, 18, 'null') % 10 = 0, NULL, substring(hex(cityHash64(n, 18)), 1, 12)) AS Nullable(String)) AS f18,
    CAST(substring(hex(cityHash64(n, 19)), 1, 12) AS Nullable(String)) AS f19,
    CAST(substring(hex(cityHash64(n, 20)), 1, 12) AS Nullable(String)) AS f20,
    CAST(concat(hex(cityHash64(n, 21)), hex(cityHash64(n, 21, 'b'))) AS Nullable(String)) AS f21,
    CAST(if(cityHash64(n, 22, 'null') % 10 = 0, NULL, concat(hex(cityHash64(n, 22)), hex(cityHash64(n, 22, 'b')))) AS Nullable(String)) AS f22,
    CAST(concat(hex(cityHash64(n, 23)), hex(cityHash64(n, 23, 'b'))) AS Nullable(String)) AS f23,
    CAST(concat(hex(cityHash64(n, 24)), hex(cityHash64(n, 24, 'b'))) AS Nullable(String)) AS f24,
    CAST(if(cityHash64(n, 25, 'null') % 10 = 0, NULL, concat(hex(cityHash64(n, 25)), hex(cityHash64(n, 25, 'b')))) AS Nullable(String)) AS f25,
    toDateTime64({sid:DateTime}, 3) + toIntervalMillisecond(n % 600000) AS event_timestamp
FROM (SELECT number AS n FROM numbers({n:UInt64} + if({v:UInt8} = 1, intDiv({n:UInt64}, 200), 0)))
WHERE NOT ({v:UInt8} = 1 AND n < {n:UInt64} AND n % 1000 < 3);

## 01-boundary.sql
-- 01-boundary.sql: a snapshot arrives in many inserts, so ingestion_timestamp cannot delimit it.
--
-- Every hour the producer starts at hh:56 and the sink writes the snapshot over 8 minutes.
-- The 09:00 snapshot lands 09:56-10:04, the 10:00 snapshot 10:56-11:04; one batch of the 10:00
-- snapshot is delivered again at 11:20 (a sink retry). ingestion_timestamp is set explicitly to
-- simulate those times; snapshot_id is what the producer stamps on every record.

TRUNCATE TABLE snapdiff.raw;
INSERT INTO snapdiff.raw (snapshot_id, record_key, f01, f02, f03, f04, f05, f06, f07, f08, f09, f10, f11, f12, f13, f14, f15, f16, f17, f18, f19, f20, f21, f22, f23, f24, f25, event_timestamp, ingestion_timestamp)
SELECT *, toDateTime64('2026-11-17 09:56:00', 3) + toIntervalMillisecond(cityHash64(record_key, snapshot_id) % 480000)
FROM snapdiff.gen(n = 10000, v = 0, sid = '2026-11-17 09:00:00');
INSERT INTO snapdiff.raw (snapshot_id, record_key, f01, f02, f03, f04, f05, f06, f07, f08, f09, f10, f11, f12, f13, f14, f15, f16, f17, f18, f19, f20, f21, f22, f23, f24, f25, event_timestamp, ingestion_timestamp)
SELECT *, toDateTime64('2026-11-17 10:56:00', 3) + toIntervalMillisecond(cityHash64(record_key, snapshot_id) % 480000)
FROM snapdiff.gen(n = 10000, v = 1, sid = '2026-11-17 10:00:00');
-- the retry: the records of the first minute of the 10:00 load, again
INSERT INTO snapdiff.raw (snapshot_id, record_key, f01, f02, f03, f04, f05, f06, f07, f08, f09, f10, f11, f12, f13, f14, f15, f16, f17, f18, f19, f20, f21, f22, f23, f24, f25, event_timestamp, ingestion_timestamp)
SELECT *, toDateTime64('2026-11-17 11:20:00', 3)
FROM snapdiff.gen(n = 10000, v = 1, sid = '2026-11-17 10:00:00')
WHERE cityHash64(record_key, snapshot_id) % 480000 < 60000;
-- what an hourly window on ingestion_timestamp holds
SELECT toStartOfHour(ingestion_timestamp) AS ingestion_hour,
       count() AS rows, uniqExact(record_key) AS records,
       groupUniqArray(toString(snapshot_id)) AS snapshots_inside
FROM snapdiff.raw
GROUP BY ingestion_hour
ORDER BY ingestion_hour;

## 02-row-hash.sql
-- 02-row-hash.sql: which hash finds the changed records.
--
-- The hash covers the 25 business fields only. record_key is the join key, ingestion_timestamp is set by the
-- sink, and event_timestamp moves on every snapshot, so none of them belongs in it.

-- the functions on literals
SELECT cityHash64(NULL::Nullable(String), 'a') AS variadic_with_null,
       cityHash64(tuple(NULL::Nullable(String), 'a')) AS tuple_with_null,
       cityHash64(tuple(''::Nullable(String), 'a')) AS tuple_with_empty,
       cityHash64(tuple('ab', 'c')) = cityHash64(tuple('a', 'bc')) AS tuple_shift_collides,
       cityHash64(concat('ab', 'c')) = cityHash64(concat('a', 'bc')) AS concat_shift_collides,
       cityHash64(ifNull(NULL::Nullable(String), '')) = cityHash64('') AS ifnull_null_equals_empty;

1. │               ᴺᵁᴸᴸ │ 6206526592849077024 │ 12190390838800561413 │                    0 │                     1 │                        1 │

## 03-diff.sql
-- 03-diff.sql: the diff of two snapshots, as rows a consumer can read.

## 04-rmv.sql
-- 04-rmv.sql: the diff as a refreshable MV that appends to snapdiff.changes, and the partial-snapshot race.
--
-- diff_rmv (guarded): compares the oldest snapshot that is not compared yet, only when the manifest row for it has
-- arrived and all its records are loaded, and writes one 'end' row per snapshot. The next refresh starts after
-- the last 'end' row, so a refresh that finds nothing to do writes nothing.
-- diff_rmv_naive: compares the newest snapshot_id in raw with the one before, whatever has arrived.

TRUNCATE TABLE snapdiff.raw;
TRUNCATE TABLE snapdiff.manifest;
TRUNCATE TABLE snapdiff.changes;
DROP VIEW IF EXISTS snapdiff.diff_rmv;
DROP VIEW IF EXISTS snapdiff.diff_rmv_naive;
DROP TABLE IF EXISTS snapdiff.changes_naive;
CREATE TABLE snapdiff.changes_naive AS snapdiff.changes;
-- snapshot 09:00: all records, then its manifest row
INSERT INTO snapdiff.raw (snapshot_id, record_key, f01, f02, f03, f04, f05, f06, f07, f08, f09, f10, f11, f12, f13, f14, f15, f16, f17, f18, f19, f20, f21, f22, f23, f24, f25, event_timestamp) SELECT * FROM snapdiff.gen(n = 10000, v = 0, sid = '2026-11-17 09:00:00');
INSERT INTO snapdiff.manifest (snapshot_id, expected_rows) SELECT '2026-11-17 09:00:00', uniqExact(record_key) FROM snapdiff.raw WHERE snapshot_id = '2026-11-17 09:00:00';
CREATE MATERIALIZED VIEW snapdiff.diff_rmv
REFRESH EVERY 1 HOUR OFFSET 30 MINUTE
APPEND TO snapdiff.changes
AS
WITH
    (SELECT max(snapshot_id) FROM snapdiff.changes WHERE change_type = 'end') AS done,
    (SELECT min(snapshot_id) FROM snapdiff.manifest WHERE snapshot_id > done) AS cur,
    (SELECT max(snapshot_id) FROM snapdiff.manifest WHERE snapshot_id < cur) AS prev,
    (SELECT any(expected_rows) FROM snapdiff.manifest WHERE snapshot_id = cur) AS cur_expected,
    (SELECT any(expected_rows) FROM snapdiff.manifest WHERE snapshot_id = prev) AS prev_expected,
    (SELECT uniqExact(record_key) FROM snapdiff.raw WHERE snapshot_id = cur) AS cur_loaded,
    (SELECT uniqExact(record_key) FROM snapdiff.raw WHERE snapshot_id = prev) AS prev_loaded,
    -- ready: an uncompared snapshot whose manifest row has arrived and whose records are all loaded,
    -- and a previous snapshot that is still complete in raw (not dropped by TTL)
    cur > done AND cur_loaded >= cur_expected AND (prev = toDateTime(0) OR prev_loaded >= prev_expected) AS ready,
    ready AND prev > toDateTime(0) AS compare
SELECT snapshot_id, prev_snapshot_id, change_type, record_key, f01, f02, f03, f04, f05, f06, f07, f08, f09, f10, f11, f12, f13, f14, f15, f16, f17, f18, f19, f20, f21, f22, f23, f24, f25
FROM
(
    -- step 2: the full rows of the changed keys only: new and changed from cur, deleted from prev
    SELECT cur AS snapshot_id, prev AS prev_snapshot_id, CAST(k.change_type AS Enum8('new' = 1, 'changed' = 2, 'deleted' = 3, 'end' = 9)) AS change_type,
           r.record_key AS record_key, r.f01, r.f02, r.f03, r.f04, r.f05, r.f06, r.f07, r.f08, r.f09, r.f10, r.f11, r.f12, r.f13, r.f14, r.f15, r.f16, r.f17, r.f18, r.f19, r.f20, r.f21, r.f22, r.f23, r.f24, r.f25
    FROM (SELECT snapshot_id, record_key, f01, f02, f03, f04, f05, f06, f07, f08, f09, f10, f11, f12, f13, f14, f15, f16, f17, f18, f19, f20, f21, f22, f23, f24, f25 FROM snapdiff.raw WHERE compare AND snapshot_id IN (prev, cur)) AS r
    INNER JOIN
    (
        -- step 1: classify every key of both snapshots by presence and row_hash (keys and hashes only)
        SELECT record_key,
               multiIf(countIf(snapshot_id = prev) = 0, 'new',
                       countIf(snapshot_id = cur) = 0, 'deleted',
                       anyIf(row_hash, snapshot_id = cur) != anyIf(row_hash, snapshot_id = prev), 'changed',
                       'same') AS change_type
        FROM snapdiff.raw
        WHERE compare AND snapshot_id IN (prev, cur)
        GROUP BY record_key
        HAVING change_type != 'same'
    ) AS k USING (record_key)
    WHERE (r.snapshot_id = cur AND k.change_type IN ('new', 'changed')) OR (r.snapshot_id = prev AND k.change_type = 'deleted')
    LIMIT 1 BY record_key
    UNION ALL
    -- one 'end' row per compared snapshot (the first one is only a baseline): where the next refresh starts,
    -- and the signal for consumers
    SELECT cur, prev, 'end', '', NULL AS f01, NULL AS f02, NULL AS f03, NULL AS f04, NULL AS f05, NULL AS f06, NULL AS f07, NULL AS f08, NULL AS f09, NULL AS f10, NULL AS f11, NULL AS f12, NULL AS f13, NULL AS f14, NULL AS f15, NULL AS f16, NULL AS f17, NULL AS f18, NULL AS f19, NULL AS f20, NULL AS f21, NULL AS f22, NULL AS f23, NULL AS f24, NULL AS f25
    WHERE ready
);

CREATE MATERIALIZED VIEW snapdiff.diff_rmv REFRESH EVERY 1 HOUR OFFSET 30 MINUTE APPEND TO snapdiff.changes AS
WITH
    (SELECT max(snapshot_id) FROM snapdiff.changes WHERE change_type = 'end') AS done,
    (SELECT min(snapshot_id) FROM snapdiff.manifest WHERE snapshot_id > done) AS cur,
    (SELECT max(snapshot_id) FROM snapdiff.manifest WHERE snapshot_id < cur) AS prev,
    (SELECT any(expected_rows) FROM snapdiff.manifest WHERE snapshot_id = cur) AS cur_expected,
    (SELECT any(expected_rows) FROM snapdiff.manifest WHERE snapshot_id = prev) AS prev_expected,
    (SELECT uniqExact(record_key) FROM snapdiff.raw WHERE snapshot_id = cur) AS cur_loaded,
    (SELECT uniqExact(record_key) FROM snapdiff.raw WHERE snapshot_id = prev) AS prev_loaded,
    -- ready: an uncompared snapshot whose manifest row has arrived and whose records are all loaded,
    -- and a previous snapshot that is still complete in raw (not dropped by TTL)
    cur > done AND cur_loaded >= cur_expected AND (prev = toDateTime(0) OR prev_loaded >= prev_expected) AS ready,
    ready AND prev > toDateTime(0) AS compare
SELECT snapshot_id, prev_snapshot_id, change_type, record_key, f01, f02, f03, f04, f05, f06, f07, f08, f09, f10, f11, f12, f13, f14, f15, f16, f17, f18, f19, f20, f21, f22, f23, f24, f25
FROM
(
    -- step 2: the full rows of the changed keys only: new and changed from cur, deleted from prev
    SELECT cur AS snapshot_id, prev AS prev_snapshot_id, CAST(k.change_type AS Enum8('new' = 1, 'changed' = 2, 'deleted' = 3, 'end' = 9)) AS change_type,
           r.record_key AS record_key, r.f01, r.f02, r.f03, r.f04, r.f05, r.f06, r.f07, r.f08, r.f09, r.f10, r.f11, r.f12, r.f13, r.f14, r.f15, r.f16, r.f17, r.f18, r.f19, r.f20, r.f21, r.f22, r.f23, r.f24, r.f25
    FROM (SELECT snapshot_id, record_key, f01, f02, f03, f04, f05, f06, f07, f08, f09, f10, f11, f12, f13, f14, f15, f16, f17, f18, f19, f20, f21, f22, f23, f24, f25 FROM snapdiff.raw WHERE compare AND snapshot_id IN (prev, cur)) AS r
    INNER JOIN
    (
        -- step 1: classify every key of both snapshots by presence and row_hash (keys and hashes only)
        SELECT record_key,
               multiIf(countIf(snapshot_id = prev) = 0, 'new',
                       countIf(snapshot_id = cur) = 0, 'deleted',
                       anyIf(row_hash, snapshot_id = cur) != anyIf(row_hash, snapshot_id = prev), 'changed',
                       'same') AS change_type
        FROM snapdiff.raw
        WHERE compare AND snapshot_id IN (prev, cur)
        GROUP BY record_key
        HAVING change_type != 'same'
    ) AS k USING (record_key)
    WHERE (r.snapshot_id = cur AND k.change_type IN ('new', 'changed')) OR (r.snapshot_id = prev AND k.change_type = 'deleted')
    LIMIT 1 BY record_key
    UNION ALL
    -- one 'end' row per compared snapshot (the first one is only a baseline): where the next refresh starts,
    -- and the signal for consumers
    SELECT cur, prev, 'end', '', NULL AS f01, NULL AS f02, NULL AS f03, NULL AS f04, NULL AS f05, NULL AS f06, NULL AS f07, NULL AS f08, NULL AS f09, NULL AS f10, NULL AS f11, NULL AS f12, NULL AS f13, NULL AS f14, NULL AS f15, NULL AS f16, NULL AS f17, NULL AS f18, NULL AS f19, NULL AS f20, NULL AS f21, NULL AS f22, NULL AS f23, NULL AS f24, NULL AS f25
    WHERE ready
);
